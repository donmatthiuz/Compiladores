.data
newline: .asciiz "\n"


.text
.globl main

main:
    li $t0, 8
    move $a0, $t0  # arg 0
    addi $sp, $sp, -4
    sw $t0, 0($sp)
    jal SumaRecursiva
    lw $t0, 0($sp)
    addi $sp, $sp, 4
    move $t1, $v0
    move $t2, $t1  # res = t5
    # print integer: res
    move $a0, $t2
    li $v0, 1  # Print integer
    syscall
    li $v0, 4  # Print newline
    la $a0, newline
    syscall
    li $v0, 10
    syscall

SumaRecursiva:
    addi $sp, $sp, -16
    sw $ra, 12($sp)
    sw $fp, 8($sp)
    move $fp, $sp
    sw $a0, 16($fp)  # param n_1
    # param None None n_1
    lw $t3, 16($fp)  # param n_1
    li $t4, 0
    slt $t5, $t4, $t3
    xori $t5, $t5, 1
    beq $t5, $zero, SumaRecursiva_L1
    li $t4, 0
    move $v0, $t4
    j SumaRecursiva_end
    j SumaRecursiva_L2
SumaRecursiva_L1:
    lw $t3, 16($fp)  # param n_1
    li $t6, 1
    sub $t7, $t3, $t6  # t2 = n_1 - 1
    move $a0, $t7  # arg 0
    addi $sp, $sp, -32
    sw $t0, 0($sp)
    sw $t1, 4($sp)
    sw $t2, 8($sp)
    sw $t3, 12($sp)
    sw $t4, 16($sp)
    sw $t5, 20($sp)
    sw $t6, 24($sp)
    sw $t7, 28($sp)
    jal SumaRecursiva
    lw $t0, 0($sp)
    lw $t1, 4($sp)
    lw $t2, 8($sp)
    lw $t3, 12($sp)
    lw $t4, 16($sp)
    lw $t5, 20($sp)
    lw $t6, 24($sp)
    lw $t7, 28($sp)
    addi $sp, $sp, 32
    move $t8, $v0
    lw $t3, 16($fp)  # param n_1
    add $t9, $t3, $t8  # t4 = n_1 + t3
    move $v0, $t9
    j SumaRecursiva_end
SumaRecursiva_L2:
SumaRecursiva_end:
    lw $ra, 12($sp)
    lw $fp, 8($sp)
    addi $sp, $sp, 16
    jr $ra
